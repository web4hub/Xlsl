cmake_minimum_required(VERSION 3.16)
project(aura_xlsl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(auraxlsl INTERFACE)
target_include_directories(auraxlsl INTERFACE engine/include)
add_executable(log_rank_test tests/log_rank_test.cpp)
target_link_libraries(log_rank_test PRIVATE auraxlsl)
enable_testing()
add_test(NAME log_rank_test COMMAND log_rank_test)
